## Quick Command Reference

```bash
# Set database connection
export DATABASE_URL="postgresql://postgres:[password]@db.[project].supabase.co:5432/postgres"

# Pre-deployment check
./supabase/migrations/20251115163857_pre_deployment_check.sh

# Create backup
pg_dump -Fc "$DATABASE_URL" > backup_$(date +%Y%m%d_%H%M%S).dump

# Deploy migration
supabase db push
# OR
psql "$DATABASE_URL" -f supabase/migrations/20251115163857_cleanup_orphaned_beach_photos.sql

# Verify success
psql "$DATABASE_URL" -c "SELECT COUNT(*) FROM beach_photos WHERE NOT EXISTS (SELECT 1 FROM beaches WHERE id = beach_photos.beach_id);"

# Check audit record
psql "$DATABASE_URL" -c "SELECT records_affected FROM data_cleanup_audit WHERE table_name = 'beach_photos' ORDER BY cleanup_date DESC LIMIT 1;"

# Rollback (if needed)
psql "$DATABASE_URL" -f supabase/migrations/20251115163857_cleanup_orphaned_beach_photos_rollback.sql
```
